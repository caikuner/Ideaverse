## 实现简单的日志上报功能

要求：支持大流量合并上传，支持分类型添加、每种类型支持配置不同的采样比例 ratio

```js

class LogReporter {
  constructor(reportUrl, interval = 60000) { // 构造函数，设置上报地址和时间间隔
    this.logBuffer = []; // 日志缓存池
    this.typeRatio = {}; // 类型采样配置
    this.reportUrl = reportUrl; // 上报接口
    this.timer = setInterval(() => this.report(), interval); // 定时上报
  }

  setTypeRatio(type, ratio) { // 设置某类型采样比例
    this.typeRatio[type] = ratio;
  }

  addLog(type, message) { // 添加日志
    const ratio = this.typeRatio[type] ?? 0;
    if (Math.random() < ratio) { // 按采样比例采集
      this.logBuffer.push({type, message, time: Date.now()});
    }
  }

  report() { // 批量上报
    if (this.logBuffer.length === 0) return;
    fetch(this.reportUrl, {
      method: 'POST',
      body: JSON.stringify(this.logBuffer),
      headers: {'Content-Type': 'application/json'}
    }).then(()=> {
        this.logBuffer.length = 0; // 上报成功清空缓存
    });
  }

  destroy() { // 销毁定时器
    clearInterval(this.timer);
  }
}

// 测试用例
const logger = new LogReporter('/api/report', 10000); // 每10秒上报
logger.setTypeRatio('error', 1);
logger.setTypeRatio('info', 0.1);
logger.addLog('error', '错误日志');
logger.addLog('info', '信息日志');
logger.setTypeRatio('custom', 0.5);
logger.addLog('custom', '自定义日志');
```

时间复杂度：添加日志 $O(1)$，上报 $O(n)$
空间复杂度：$O(n)$
实现说明：支持定时批量上报、类型采样和动态配置。
可优化点：支持本地存储持久化、异常重试、上报量动态调整等。

### 重试机制

指数退避 (Exponential Backoff) 是一种在网络请求或重试操作中常用的算法，它通过逐步增加重试间隔时间来减轻服务器压力和提高请求成功率。

指数退避的核心思想是：
1. 第一次失败后等待基础时间
2. 每次重试后等待时间呈指数增长
3. 通常设置最大重试次数和最大等待时间

#### 简单实现示例

```javascript
async function fetchWithRetry(url, options = {}, maxRetries = 5, baseDelay = 1000, maxDelay = 30000) {
    let attempt = 0;
    
    while (attempt < maxRetries) {
        try {
            const response = await fetch(url, options);
            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            return await response.json();
        } catch (error) {
            attempt++;
            if (attempt >= maxRetries) throw error;
            
            // 计算指数退避延迟时间
            const delay = Math.min(baseDelay * Math.pow(2, attempt - 1), maxDelay); // 最大30秒
            console.log(`请求失败，${delay}ms后重试 (${attempt}/${maxRetries})`, error);
            
            // 等待延迟时间
            await new Promise(resolve => setTimeout(resolve, delay));
        }
    }
}
```

#### 带抖动 (Jitter) 的高级实现

为了防止多个客户端同步重试造成 " 惊群效应 "，可以添加随机抖动：

```javascript
async function fetchWithExponentialBackoff(
    url,
    options = {},
    maxRetries = 5,
    baseDelay = 1000,
    maxDelay = 30000
) {
    let attempt = 0;
    
    while (true) {
        try {
            const response = await fetch(url, options);
            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            return await response.json();
        } catch (error) {
            attempt++;
            if (attempt >= maxRetries) throw error;
            
            // 计算指数退避时间并添加随机抖动
            const delay = Math.min(
                baseDelay * Math.pow(2, attempt - 1) + 
                Math.random() * baseDelay, // 随机抖动
                maxDelay
            );
            
            console.log(`请求失败，${Math.round(delay)}ms后重试 (${attempt}/${maxRetries})`);
            await new Promise(resolve => setTimeout(resolve, delay));
        }
    }
}
```

#### 使用示例

```javascript
// 使用示例
fetchWithRetry('https://api.example.com/data')
    .then(data => console.log('成功获取数据:', data))
    .catch(error => console.error('最终失败:', error));
```

#### 封装为可重用类

```javascript
class ExponentialBackoff {
    constructor({
        maxRetries = 5,
        baseDelay = 1000,
        maxDelay = 30000,
        jitter = true
    } = {}) {
        this.maxRetries = maxRetries;
        this.baseDelay = baseDelay;
        this.maxDelay = maxDelay;
        this.jitter = jitter;
    }
    
    async execute(requestFn) {
        let attempt = 0;
        
        while (true) {
            try {
                return await requestFn();
            } catch (error) {
                attempt++;
                if (attempt >= this.maxRetries) throw error;
                
                let delay = Math.min(
                    this.baseDelay * Math.pow(2, attempt - 1),
                    this.maxDelay
                );
                
                if (this.jitter) {
                    delay += Math.random() * this.baseDelay;
                }
                
                await new Promise(resolve => setTimeout(resolve, delay));
            }
        }
    }
}

// 使用示例
const backoff = new ExponentialBackoff({ maxRetries: 3 });
backoff.execute(() => fetch('https://api.example.com/data'))
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error('最终失败:', error));
```

#### 适用场景

1. API 请求失败重试
2. 数据库连接重试
3. WebSocket 连接重连
4. 任何可能因临时故障而失败的操作

指数退避算法能有效避免因频繁重试导致的服务器过载问题，特别是在分布式系统中非常有用。
