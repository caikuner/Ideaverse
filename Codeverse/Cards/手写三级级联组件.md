## React 实现方案

### 1. 组件结构设计

```jsx
import React, { useState, useEffect } from 'react';

const ThreeLevelCascader = ({ data, onChange }) => {
  const [selected, setSelected] = useState({
    level1: null,
    level2: null,
    level3: null
  });

  const [options, setOptions] = useState({
    level1: [],
    level2: [],
    level3: []
  });

  // 初始化一级选项
  useEffect(() => {
    setOptions(prev => ({
      …prev,
      level1: data
    }));
  }, [data]);

  // 一级选择变化时更新二级选项
  useEffect(() => {
    if (selected.level1) {
      setOptions(prev => ({
        …prev,
        level2: selected.level1.children || [],
        level3: [] // 清空三级选项
      }));
      setSelected(prev => ({
        …prev,
        level2: null,
        level3: null
      }));
    }
  }, [selected.level1]);

  // 二级选择变化时更新三级选项
  useEffect(() => {
    if (selected.level2) {
      setOptions(prev => ({
        …prev,
        level3: selected.level2.children || []
      }));
      setSelected(prev => ({
        …prev,
        level3: null
      }));
    }
  }, [selected.level2]);

  // 选择变化回调
  const handleChange = (level, value) => {
    const newSelected = {
      …selected,
      [level]: value
    };
    setSelected(newSelected);
    if (onChange) {
      onChange(newSelected);
    }
  };

  return (
    <div className="cascader-container">
      <select
        value={selected.level1?.value || ''}
        onChange={(e) => handleChange(
          'level1',
          options.level1.find(item => item.value === e.target.value) || null
        )}
      >
        <option value="">请选择一级</option>
        {options.level1.map(item => (
          <option key={item.value} value={item.value}>
            {item.label}
          </option>
        ))}
      </select>

      <select
        value={selected.level2?.value || ''}
        onChange={(e) => handleChange(
          'level2',
          options.level2.find(item => item.value === e.target.value) || null
        )}
        disabled={!selected.level1}
      >
        <option value="">请选择二级</option>
        {options.level2.map(item => (
          <option key={item.value} value={item.value}>
            {item.label}
          </option>
        ))}
      </select>

      <select
        value={selected.level3?.value || ''}
        onChange={(e) => handleChange(
          'level3',
          options.level3.find(item => item.value === e.target.value) || null
        )}
        disabled={!selected.level2}
      >
        <option value="">请选择三级</option>
        {options.level3.map(item => (
          <option key={item.value} value={item.value}>
            {item.label}
          </option>
        ))}
      </select>
    </div>
  );
};

export default ThreeLevelCascader;
```

## Vue 实现方案

### 1. 组件实现

```vue
<template>
  <div class="cascader-container">
    <select v-model="selected.level1" @change="handleLevel1Change">
      <option :value="null">请选择一级</option>
      <option 
        v-for="item in options.level1" 
        :key="item.value" 
        :value="item"
      >
        {{ item.label }}
      </option>
    </select>

    <select 
      v-model="selected.level2" 
      @change="handleLevel2Change"
      :disabled="!selected.level1"
    >
      <option :value="null">请选择二级</option>
      <option 
        v-for="item in options.level2" 
        :key="item.value" 
        :value="item"
      >
        {{ item.label }}
      </option>
    </select>

    <select 
      v-model="selected.level3" 
      @change="handleLevel3Change"
      :disabled="!selected.level2"
    >
      <option :value="null">请选择三级</option>
      <option 
        v-for="item in options.level3" 
        :key="item.value" 
        :value="item"
      >
        {{ item.label }}
      </option>
    </select>
  </div>
</template>

<script>
export default {
  name: 'ThreeLevelCascader',
  props: {
    data: {
      type: Array,
      required: true
    }
  },
  data() {
    return {
      selected: {
        level1: null,
        level2: null,
        level3: null
      },
      options: {
        level1: [],
        level2: [],
        level3: []
      }
    };
  },
  watch: {
    data: {
      immediate: true,
      handler(newVal) {
        this.options.level1 = newVal;
      }
    },
    'selected.level1'(newVal) {
      this.options.level2 = newVal ? newVal.children || [] : [];
      this.selected.level2 = null;
      this.selected.level3 = null;
    },
    'selected.level2'(newVal) {
      this.options.level3 = newVal ? newVal.children || [] : [];
      this.selected.level3 = null;
    },
    selected: {
      deep: true,
      handler(newVal) {
        this.$emit('change', newVal);
      }
    }
  },
  methods: {
    handleLevel1Change() {
      this.selected.level2 = null;
      this.selected.level3 = null;
    },
    handleLevel2Change() {
      this.selected.level3 = null;
    },
    handleLevel3Change() {
      // 三级选择变化无需额外处理
    }
  }
};
</script>

<style scoped>
.cascader-container select {
  margin-right: 10px;
  padding: 5px;
}
</style>
```

## 关键实现点说明

1. **数据流控制**：
   - 一级选择变化时清空并更新二级选项
   - 二级选择变化时清空并更新三级选项

2. **状态管理**：
   - 使用 `selected` 对象存储当前各级选中项
   - 使用 `options` 对象存储当前可选项

3. **UI 交互优化**：
   - 未选择上级时禁用下级选择器
   - 每次选择变化后清空下级选择

4. **数据结构要求**：

   ```typescript
   interface CascaderOption {
     value: string;
     label: string;
     children?: CascaderOption[];
   }
   ```

3. **样式美化**：

   - 添加动画效果

4. **可配置化**：
   - 通过 props 控制显示层级数
   - 自定义选项渲染模板

这个实现保持了简洁性同时提供了完整的级联功能，可以根据实际需求进一步扩展。
