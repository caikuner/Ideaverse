在 JavaScript 中，我们经常需要在扁平数组和树形结构之间进行转换，比如菜单树、权限树。

> 手写版：[[nodeArrayToTree.js]] [[nodeTreeToArray.js]]

### 1. 数组转树（扁平数组 → 树形结构）

假设我们有一个扁平数组，每个元素包含 id 和 parentId 属性：

```javascript
const flatArray = [
  { id: 1, name: "节点1", parentId: null },
  { id: 2, name: "节点2", parentId: 1 },
  { id: 3, name: "节点3", parentId: 1 },
  { id: 4, name: "节点4", parentId: 2 },
  { id: 5, name: "节点5", parentId: null },
  { id: 6, name: "节点6", parentId: 5 },
];
```

- 分析入参与出参

```ts
interface TreeNode {
  id: number;
  name: string;
  children?: TreeNode[];
}

interface ListItem extends Extract<TreeNode, "children"> {
  parentId?: number;
}
```

#### 方法一：递归实现

```javascript
function arrayToTree(items, parentId = null) {
  return items
    .filter((item) => item.parentId === parentId)
    .map((item) => ({
      ...item,
      children: arrayToTree(items, item.id),
    }));
}

const tree = arrayToTree(flatArray);
console.log(tree);
```

#### 方法二：非递归实现（使用对象映射）

```javascript
function arrayToTree(items) {
  const result = [];
  const itemMap = {};

  // 先转成map存储
  for (const item of items) {
    itemMap[item.id] = { ...item, children: [] };
  }

  for (const item of items) {
    const { id, parentId } = item;
    const treeItem = treeMap[id];

    if (parentId === null) {
      result.push(treeItem);
    } else {
      if (!itemMap[parentId]) {
        itemMap[parentId] = { children: [] };
      }
      itemMap[parentId].children.push(treeItem);
    }
  }

  return result;
}

const tree = arrayToTree(flatArray);
console.log(tree);
```

### 2. 树转数组（树形结构 → 扁平数组）

假设我们有一个树形结构：

```javascript
const tree = [
  {
    id: 1,
    name: "节点1",
    children: [
      {
        id: 2,
        name: "节点2",
        children: [{ id: 4, name: "节点4", children: [] }],
      },
      { id: 3, name: "节点3", children: [] },
    ],
  },
  {
    id: 5,
    name: "节点5",
    children: [{ id: 6, name: "节点6", children: [] }],
  },
];
```

#### 方法一：递归实现

```javascript
function treeToArray(tree, parentId = null) {
  return tree.reduce((acc, node) => {
    const { children, ...rest } = node;
    acc.push({ ...rest, parentId });
    if (children && children.length) {
      acc.push(...treeToArray(children, node.id));
    }
    return acc;
  }, []);
}

const flatArray = treeToArray(tree);
console.log(flatArray);
```

#### 方法二：非递归实现（使用栈）

```javascript
function treeToArray(tree) {
  const result = [];
  const stack = [...tree];

  while (stack.length) {
    const node = stack.pop();
    const { children, ...rest } = node;

    result.push(rest);

    if (children && children.length) {
      // 从右到左压入栈，保证处理顺序正确
      for (let i = children.length - 1; i >= 0; i--) {
        stack.push({
          ...children[i],
          parentId: node.id,
        });
      }
    }
  }

  return result;
}

const flatArray = treeToArray(tree);
console.log(flatArray);
```

### 3. 性能考虑

- 对于大型数据集，非递归实现通常性能更好，因为递归可能导致堆栈溢出
- 对象映射方法在数组转树时效率较高，时间复杂度接近 O(n)
