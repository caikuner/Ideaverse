```

git init // 新建 git 代码库

git add // 添加指定文件到暂存区

git rm // 删除工作区文件，并且将这次删除放入暂存区

git commit -m [message] // 提交暂存区到仓库区

git branch // 列出所有分支

git checkout -b [branch] // 新建一个分支，并切换到该分支

git status // 显示有变更的文件

```
