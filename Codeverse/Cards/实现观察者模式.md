### 观察者模式

```
[Subject] ←-------→ [Observer]
| ↑                 ↑
└── notify()────────┘
```

---

### 实现

```js
// 基类
class Subject {
  constructor() {
    this.observers = [];
  }

  // 外部订阅者调用，用于注册和取消订阅
  subscribe(observer) {
    this.observers.push(observer);
    return () => unsubscribe(observer);
  }

  unsubscribe(observer) {
    this.observers = this.observers.filter((obs) => obs !== observer);
  }

  // 通知所有外部订阅者更新
  notify(data) {
    // 立即触发所有观察者
    this.observers.forEach((observer) => observer.update(data));
  }
}

// 观察者接口
class Observer {
  update(data) {
    throw new Error("必须 update 方法, 实现更新逻辑");
  }
}
```

```js
// 实现示例: 股票市场价格监控
class StockMarket extends Subject {
  constructor(price) {
    super();
    this.stockPrice = price;
  }
  changePrice(newPrice) {
    if (newPrice === this.stockPrice) return;

    this.stockPrice = newPrice;
    this.notify({ price: newPrice });
  }
}

class EmailAlert extends Observer {
  update({ price }) {
    console.log("股价改变了！");
    if (price > 150) {
      console.error("发送邮件预警：股价过高！");
    }
  }
}

// 使用示例
const stockMarket = new StockMarket(100);
const emailAlert = new EmailAlert();

stockMarket.subscribe(emailAlert);
stockMarket.changePrice(100); // 已订阅，股价未改变，不更新
stockMarket.changePrice(200); // 已订阅，做出更新

stockMarket.unsubscribe(emailAlert);
stockMarket.changePrice(300); // 已取消订阅，不再更新
```
