> Q: 如何做全局的请求处理，比如统一处理登录态、全局错误处理?
> 你是如何封装请求库的

在前端开发中，封装了请求库是实现全局请求处理，包含统一处理登录态、全局错误处理等功能。

## 核心功能设计

1. **统一配置**：基础 URL、超时时间、请求头等
2. **登录态管理**：自动携带 Token、Token 过期处理
3. **全局错误处理**：HTTP 错误、业务错误、网络错误
4. **拦截器机制**：请求/响应拦截器
5. **请求取消**：避免重复请求
6. **加载状态**：全局请求 loading 控制

## 代码实现

### axios 封装

```jsx
// src/utils/request.js
import axios from "axios";
import { message, Modal } from "antd";
import store from "@/store";
import router from "@/router";

// 创建axios实例
const service = axios.create({
  baseURL: process.env.REACT_APP_BASE_API,
  timeout: 15000,
  withCredentials: true, // 跨域请求时发送cookies
});

// 请求队列 - 用于取消重复请求
const pendingRequests = new Map();

// 生成请求唯一标识
const generateReqKey = (config) => {
  const { method, url, params, data } = config;
  return [method, url, JSON.stringify(params), JSON.stringify(data)].join("&");
};

// 添加请求到队列
const addPendingRequest = (config) => {
  const requestKey = generateReqKey(config);
  config.cancelToken =
    config.cancelToken ||
    new axios.CancelToken((cancel) => {
      if (!pendingRequests.has(requestKey)) {
        pendingRequests.set(requestKey, cancel);
      }
    });
};

// 移除队列中的请求
const removePendingRequest = (config) => {
  const requestKey = generateReqKey(config);
  if (pendingRequests.has(requestKey)) {
    const cancel = pendingRequests.get(requestKey);
    cancel(requestKey);
    pendingRequests.delete(requestKey);
  }
};

// 请求拦截器
service.interceptors.request.use(
  (config) => {
    // 取消重复请求
    removePendingRequest(config);
    addPendingRequest(config);

    // 全局loading状态
    store.dispatch({ type: "SET_LOADING", payload: true });

    // 添加token到请求头
    const token = localStorage.getItem("access_token");
    if (token) {
      config.headers["Authorization"] = `Bearer ${token}`;
    }

    // 特殊处理文件上传
    if (config.isUpload) {
      config.headers["Content-Type"] = "multipart/form-data";
    }

    return config;
  },
  (error) => {
    store.dispatch({ type: "SET_LOADING", payload: false });
    return Promise.reject(error);
  },
);

// 响应拦截器
service.interceptors.response.use(
  (response) => {
    // 请求完成后移除队列
    removePendingRequest(response.config);
    store.dispatch({ type: "SET_LOADING", payload: false });

    // 统一处理响应格式
    const res = response.data;

    // 业务成功状态码
    if (res.code === 0 || res.success) {
      return res.data || res.result;
    }

    // 业务错误处理
    handleBusinessError(res);
    return Promise.reject(new Error(res.message || "业务错误"));
  },
  (error) => {
    // 请求完成后移除队列
    error.config && removePendingRequest(error.config);
    store.dispatch({ type: "SET_LOADING", payload: false });

    // 全局错误处理
    handleHttpError(error);
    return Promise.reject(error);
  },
);

// 处理HTTP错误
function handleHttpError(error) {
  if (axios.isCancel(error)) {
    // 请求被取消
    console.log("请求取消:", error.message);
    return;
  }

  if (!error.response) {
    // 网络错误
    message.error("网络连接异常，请检查网络设置");
    return;
  }

  const status = error.response.status;

  switch (status) {
    case 401:
      handleUnauthorized();
      break;
    case 403:
      message.error("拒绝访问，权限不足");
      break;
    case 404:
      message.error("请求的资源不存在");
      break;
    case 500:
      message.error("服务器内部错误");
      break;
    case 503:
      message.error("服务不可用");
      break;
    default:
      message.error(`请求错误: ${status}`);
  }
}

// 处理业务错误
function handleBusinessError(response) {
  const { code, message: msg } = response;

  // 根据业务错误码进行不同处理
  switch (code) {
    case 1001: // Token过期
      handleUnauthorized();
      break;
    case 1002: // 权限不足
      message.error("权限不足，无法操作");
      break;
    case 1003: // 参数错误
      message.error(msg || "请求参数错误");
      break;
    default:
      message.error(msg || "业务处理异常");
  }
}

// 处理未授权情况
function handleUnauthorized() {
  // 清除用户信息
  localStorage.removeItem("access_token");
  localStorage.removeItem("user_info");

  // 提示用户
  Modal.confirm({
    title: "登录状态已过期",
    content: "请重新登录",
    okText: "重新登录",
    cancelText: "取消",
    onOk() {
      // 跳转到登录页
      router.replace("/login");
      // 重置用户状态
      store.dispatch({ type: "RESET_USER" });
    },
    onCancel() {
      console.log("取消重新登录");
    },
  });
}

// 封装请求方法
const request = {
  get(url, params, config = {}) {
    return service.get(url, { params, ...config });
  },

  post(url, data, config = {}) {
    return service.post(url, data, config);
  },

  put(url, data, config = {}) {
    return service.put(url, data, config);
  },

  delete(url, params, config = {}) {
    return service.delete(url, { params, ...config });
  },

  upload(url, file, config = {}) {
    const formData = new FormData();
    formData.append("file", file);
    return service.post(url, formData, {
      ...config,
      isUpload: true,
    });
  },

  // 取消所有请求
  cancelAllRequests() {
    pendingRequests.forEach((cancel, key) => {
      cancel(key);
    });
    pendingRequests.clear();
  },

  // 设置认证token
  setAuthorization(token) {
    localStorage.setItem("access_token", token);
  },

  // 清除认证token
  clearAuthorization() {
    localStorage.removeItem("access_token");
  },
};

export default request;
```

## 使用示例

```jsx
// src/api/user.js
import request from "@/utils/request";

export const getUserInfo = (userId) => {
  return request.get("/user/info", { id: userId });
};

export const updateUserInfo = (userData) => {
  return request.put("/user/update", userData);
};

export const uploadAvatar = (file) => {
  return request.upload("/user/avatar", file);
};
```

```jsx
// src/components/UserProfile.js
import React, { useEffect, useState } from 'react';
import { getUserInfo } from '@/api/user';
import { useRequest } from '@/hooks'

const UserProfile = ({ userId }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
	const {loading, data, error} = await useRequest(
		() => getUserInfo(userId)
	)
  }, [userId]);

  if (loading) return <div>加载中...</div>;
  if (error || !user) return <div>暂无数据</div>;

  return (
    <div>
      <h2>{user.name}</h2>
      <p>邮箱: {user.email}</p>
      <p>角色: {user.role}</p>
    </div>
  );
};

export default UserProfile;
```

## 设计亮点

1. **多维度错误处理**：
   - HTTP 状态码错误
   - 业务逻辑错误
   - 网络错误
   - 请求取消处理

2. **登录态自动管理**：
   - 自动添加 Token 到请求头
   - Token 过期自动跳转登录页
   - 统一清除用户状态

3. **请求防重处理**：
   - 使用 Map 存储请求标识
   - 自动取消重复请求
   - 提供手动取消所有请求的方法

4. **全局状态集成**：
   - 自动管理全局 loading 状态
   - 与 Redux 状态管理集成
   - 与路由系统联动

5. **文件上传优化**：
   - 特殊处理 Content-Type
   - 简化文件上传接口

## 最佳实践建议

1. **环境区分**：

   ```js
   // .env.development
   REACT_APP_BASE_API = "http://dev-api.example.com";

   // .env.production
   REACT_APP_BASE_API = "https://api.example.com";
   ```

2. **错误码统一管理**：

   ```js
   // src/constants/errorCode.js
   export const ERROR_CODES = {
     TOKEN_EXPIRED: 1001,
     PERMISSION_DENIED: 1002,
     PARAMS_ERROR: 1003,
     // ...
   };
   ```

3. **请求重试机制**（可选增强）：

   ```js
   // 在响应拦截器中添加重试逻辑
   if (error.config && error.response?.status === 503) {
     const retryCount = error.config.retryCount || 0;
     if (retryCount < 3) {
       error.config.retryCount = retryCount + 1;
       return service(error.config);
     }
   }
   ```

4. **性能监控**：

   ```js
   // 在请求拦截器中记录开始时间
   config.metadata = { startTime: Date.now() };

   // 在响应拦截器中计算耗时
   const endTime = Date.now();
   const duration = endTime - config.metadata.startTime;
   console.log(`请求 ${config.url} 耗时: ${duration}ms`);
   ```

这个封装方案提供了全面的全局请求处理能力，能够满足企业级应用的需求，同时保持了良好的扩展性和可维护性。
