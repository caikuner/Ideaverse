> Q: 实现插入排序

插入排序就是洗牌法，把新牌插入之前排好序的旧牌之中。从后往前逐个比较，找到插入位置

```js
function insertionSort(arr) {
  if (arr.length <= 1) return arr;

  for (let i = 1; i < arr.length; i++) {
    // 当前要插入的元素
    const current = arr[i];
    // 已排序部分的最后一个元素索引
    let j = i - 1;

    // 在已排序部分从后向前扫描
    while (j >= 0 && arr[j] > current) {
      // 如果已排序元素大于当前元素，则向后移动
      arr[j + 1] = arr[j];
      j--;
    }

    // 插入当前元素到正确位置(已排序部分的后面)
    arr[j + 1] = current;
  }
  return arr;
}
```
