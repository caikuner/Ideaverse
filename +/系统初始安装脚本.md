- ubuntu

```bash
mkdir -p ~/backups/
cd backups

# 镜像源
# https://mirrors.tuna.tsinghua.edu.cn/help/ubuntu/    >  sources.list
touch sources.list
# ... copy link content above
mkdir apt
cp /etc/apt/sources.list ./apt/sources.list.old
sudo cp sources.list /etc/apt/sources.list
mv sources.list ./apt


# update
sudo apt update && sudo apt upgrade

# base util
sudo apt install git curl wget make gcc fontconfig net-tools iputils-ping zip unzip apt-show-versions

# font
# https://www.nerdfonts.com/font-downloads

unzip xxx.zip
mv *.ttf /usr/local/share/fonts/
fc-list
fc-cache -fv


# zsh
sudo apt install zsh
sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)"

# neovim
sudo apt install neovim
nvim --version  # 版本很低
sudo apt-get install software-properties-common

sudo add-apt-repository ppa:neovim-ppa/stable # 稳定版
sudo apt-add-repository ppa:neovim-ppa/unstable # 开发板
sudo apt-get update
sudo apt-get install neovim

git clone https://github.com/caikuner/dotfiles-nvim.git "${XDG_CONFIG_HOME:-$HOME/.config}"/nvim

cd ~/.config/nvim
git remote add upstream https://github.com/nvim-lua/kickstart.nvim.git
git pull upstream master


# ripgrep, lazygit, ranger, bat, mycli
sudo apt install ripgrep

# lazygit
LAZYGIT_VERSION=$(curl -s "https://api.github.com/repos/jesseduffield/lazygit/releases/latest" | grep -Po '"tag_name": "v\K[^"]*')
curl -Lo lazygit.tar.gz "https://github.com/jesseduffield/lazygit/releases/latest/download/lazygit_${LAZYGIT_VERSION}_Linux_x86_64.tar.gz"
tar xf lazygit.tar.gz lazygit
sudo install lazygit /usr/local/bin

# pip
pip config set global.index-url https://pypi.tuna.tsinghua.edu.cn/simple

```
